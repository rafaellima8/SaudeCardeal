#!/usr/bin/env python3
"""
download_vigilancia_sinan.py

Uso:
  python download_vigilancia_sinan.py --url "http://vigilancia.saude.mg.gov.br/index.php/sistemas-de-informacao/agravos-de-notificacao-sinan/" 

O script:
- Faz GET na página inicial;
- Extrai links para /index.php/download/... e quaisquer .pdf encontrados;
- Normaliza URLs;
- Faz download com retry/backoff;
- Salva arquivos em ./downloads/vigilancia_mg/ mantendo nome original;
- Gera download_report.json e downloader.log.
"""

import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin, urlparse
import time, os, sys, argparse, json, logging
from pathlib import Path
from requests.adapters import HTTPAdapter, Retry

# -------- CONFIG --------
DEFAULT_URL = "http://vigilancia.saude.mg.gov.br/index.php/sistemas-de-informacao/agravos-de-notificacao-sinan/"
OUT_DIR = Path("./downloads/vigilancia_mg")
USER_AGENT = "Mozilla/5.0 (compatible; VigilanciaDownloader/1.0; +https://example.org)"
TIMEOUT = 20
MAX_RETRIES = 5
BACKOFF_FACTOR = 1.0
# ------------------------

logging.basicConfig(
    filename="downloader.log",
    level=logging.INFO,
    format="%(asctime)s %(levelname)s %(message)s",
)
console = logging.StreamHandler()
console.setLevel(logging.INFO)
logging.getLogger("").addHandler(console)

def build_session():
    s = requests.Session()
    retries = Retry(total=MAX_RETRIES, backoff_factor=BACKOFF_FACTOR,
                    status_forcelist=[429,500,502,503,504], allowed_methods=["GET","POST"])
    s.mount("https://", HTTPAdapter(max_retries=retries))
    s.mount("http://", HTTPAdapter(max_retries=retries))
    s.headers.update({"User-Agent": USER_AGENT})
    return s

def fetch_html(session, url):
    logging.info(f"GET {url}")
    r = session.get(url, timeout=TIMEOUT)
    r.raise_for_status()
    return r.text

def parse_download_links(base_url, html):
    soup = BeautifulSoup(html, "html.parser")
    links = set()
    # collect anchor hrefs
    for a in soup.find_all("a", href=True):
        href = a["href"].strip()
        # pick right-away PDFs or index.php/download/... pages
        if href.lower().endswith(".pdf"):
            full = urljoin(base_url, href)
            links.add(full)
        elif "/index.php/download" in href or "/download/" in href:
            full = urljoin(base_url, href)
            links.add(full)
    return sorted(links)

def extract_pdf_from_download_page(session, page_url):
    """If page_url is a download page (HTML), look for direct pdf link(s)."""
    try:
        html = fetch_html(session, page_url)
    except Exception as e:
        logging.warning(f"Failed to fetch download page {page_url}: {e}")
        return []
    soup = BeautifulSoup(html, "html.parser")
    pdfs = []
    for a in soup.find_all("a", href=True):
        href = a["href"].strip()
        if href.lower().endswith(".pdf"):
            pdfs.append(urljoin(page_url, href))
    # Some pages have direct link text or button with query param wpdmdl=...
    # Also check for wp-content/uploads direct links
    for tag in soup.select("a"):
        href = tag.get("href", "")
        if "wp-content/uploads" in href and href.lower().endswith(".pdf"):
            pdfs.append(urljoin(page_url, href))
    return sorted(set(pdfs))

def safe_filename(url):
    p = urlparse(url)
    name = os.path.basename(p.path)
    if not name:
        # fallback to last query param or hashed name
        name = p.query.replace("&", "_").replace("=", "_") or "file.pdf"
    return name

def download_file(session, url, out_dir: Path):
    out_dir.mkdir(parents=True, exist_ok=True)
    filename = safe_filename(url)
    dest = out_dir / filename
    if dest.exists() and dest.stat().st_size > 0:
        logging.info(f"Already exists, skipping: {dest}")
        return {"url": url, "path": str(dest), "status": "skipped"}
    try:
        with session.get(url, stream=True, timeout=TIMEOUT) as r:
            r.raise_for_status()
            # attempt to determine filename from headers if different
            cd = r.headers.get("content-disposition")
            if cd and "filename" in cd:
                # parse filename
                import re
                m = re.search(r'filename\*=UTF-8\'\'(.+)$', cd)
                if m:
                    filename2 = m.group(1)
                else:
                    m2 = re.search(r'filename="?([^"]+)"?', cd)
                    filename2 = m2.group(1) if m2 else filename
                dest = out_dir / filename2
            # stream to file
            tmp = dest.with_suffix(dest.suffix + ".part")
            with open(tmp, "wb") as f:
                for chunk in r.iter_content(chunk_size=8192):
                    if chunk:
                        f.write(chunk)
            tmp.rename(dest)
            logging.info(f"Downloaded: {url} -> {dest}")
            return {"url": url, "path": str(dest), "status": "ok"}
    except Exception as e:
        logging.error(f"Error downloading {url}: {e}")
        return {"url": url, "path": None, "status": f"error: {e}"}

def crawl_and_download(base_url, out_dir, dry_run=False):
    session = build_session()
    results = {"base_url": base_url, "discovered": [], "pdfs": [], "errors": []}
    try:
        html = fetch_html(session, base_url)
    except Exception as e:
        msg = f"Failed to fetch base URL {base_url}: {e}"
        logging.error(msg)
        results["errors"].append(msg)
        return results

    links = parse_download_links(base_url, html)
    logging.info(f"Found {len(links)} candidate links (download pages or direct pdfs).")
    results["discovered"] = links

    # for each candidate, if it's a PDF -> download; if it's a download page -> extract pdf links
    pdf_urls = set()
    for link in links:
        if link.lower().endswith(".pdf"):
            pdf_urls.add(link)
        else:
            # try to extract from page
            try:
                found = extract_pdf_from_download_page(session, link)
                if found:
                    logging.info(f"From page {link} found {len(found)} pdf(s).")
                    pdf_urls.update(found)
                else:
                    logging.info(f"No PDF found on page {link}.")
            except Exception as e:
                logging.warning(f"Error extracting from {link}: {e}")
                results["errors"].append(f"extract:{link}:{e}")

    logging.info(f"Total distinct PDF links to download: {len(pdf_urls)}")

    # Optionally: try to follow more pagination (not implemented by default)

    if dry_run:
        results["pdfs"] = [{"url": u, "status": "dry-run"} for u in sorted(pdf_urls)]
        return results

    # download each
    for u in sorted(pdf_urls):
        r = download_file(session, u, Path(out_dir))
        results["pdfs"].append(r)
        # small pause to be polite
        time.sleep(0.5)

    # write report
    report_path = Path(out_dir) / "download_report.json"
    with open(report_path, "w", encoding="utf-8") as f:
        json.dump(results, f, ensure_ascii=False, indent=2)

    logging.info(f"Download finished; report at {report_path}")
    return results

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Downloader for Vigilancia MG SINAN fichas")
    parser.add_argument("--url", "-u", default=DEFAULT_URL, help="Página base com lista de fichas")
    parser.add_argument("--out", "-o", default=str(OUT_DIR), help="Diretório de saída")
    parser.add_argument("--dry-run", action="store_true", help="Somente checar links, não baixar")
    args = parser.parse_args()

    res = crawl_and_download(args.url, args.out, dry_run=args.dry_run)
    print("Done. Report:")
    print(json.dumps(res, ensure_ascii=False, indent=2))
