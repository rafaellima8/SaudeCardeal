```typescript
// paciente_cpf ← cpf
// paciente_cns ← cns
// paciente_sexo ← gender
// paciente_mae ← motherName
// res_logradouro ← address
// res_bairro ← neighborhood
// res_municipio ← city
// res_uf ← state
```

