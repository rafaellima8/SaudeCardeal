import React from 'react'
import Input from '@/components/ui/Input'
import Select from '@/components/ui/Select'
import DateInput from '@/components/ui/DateInput'
import { SinanTemplate } from 'shared/sinan/template-registry'

export default function DynamicForm({ schema, onSubmit }) {
  const [values, setValues] = React.useState({})

  function renderField(f) {
    const common = { key: f.name, label: f.label, required: f.required }
    switch(f.type) {
      case 'string': return <Input {...common} value={values[f.name]||''} onChange={(v)=>setValues(s=>({...s,[f.name]:v}))} />
      case 'date': return <DateInput {...common} value={values[f.name]||''} onChange={(v)=>setValues(s=>({...s,[f.name]:v}))} />
      case 'select': return <Select {...common} options={f.options} value={values[f.name]||''} onChange={(v)=>setValues(s=>({...s,[f.name]:v}))} />
      // ...
    }
  }

  return (
    <form onSubmit={(e)=>{e.preventDefault(); onSubmit(values)}}>
      {schema.fields.map(renderField)}
      <button type="submit">Enviar</button>
    </form>
  )
}
